'; grid.appendChild(card); var meta5 = Array(p.dias.length).fill(5); destruir(canvasId); graf[canvasId] = new Chart(document.getElementById(canvasId), { type:'line', data:{labels:diasLabels.slice(0, p.dias.length),datasets:[ {label:p.vendedor,data:p.dias,borderColor:cor,backgroundColor:'rgba(' + hexToRgb(cor) + ',0.1)',fill:false,tension:0.3,borderWidth:2,pointRadius:2}, {label:'Meta 5/dia',data:meta5,borderColor:'#FF5C5C',borderDash:[5,5],fill:false,tension:0,borderWidth:1,pointRadius:0} ]}, options:{responsive:true,maintainAspectRatio:false,plugins:{legend:{display:false}},scales:{y:{beginAtZero:true,ticks:{color:'#8CA0B3',font:{size:9}},max:10},x:{display:false}}} }); }); } function hexToRgb(hex) { var r = parseInt(hex.substr(1,2),16); var g = parseInt(hex.substr(3,2),16); var b = parseInt(hex.substr(5,2),16); return r + ',' + g + ',' + b; }
